<script>
  // ====================================================================
  // KONFIGURASI — WAJIB DIISI sebelum file ini di-upload ke hosting.
  // ====================================================================
  // API_URL: URL Web App Apps Script kamu, dari Deploy > Manage deployments
  //          (yang berakhiran /exec), CONTOH:
  //          'https://script.google.com/macros/s/AKfycb.../exec'
  // API_TOKEN: harus SAMA PERSIS dengan nilai CONFIG.API_TOKEN di Code.gs
  const API_URL = 'https://script.google.com/a/macros/murup.id/s/AKfycbxoKpFQGcswc_6iXoag8EeDJAdob6uUQcpZND8RJgVbqvuxa2BlQAhmPuk2myp31rxNTw/exec';
  const API_TOKEN = 'Inventory_Alat_Murup123!';
  // ====================================================================

  if (API_URL.indexOf('PASTE_URL') !== -1 || API_TOKEN.indexOf('GANTI-DENGAN') !== -1) {
    document.getElementById('config-warning').classList.remove('hidden');
  }

  /**
   * Panggil backend Apps Script lewat fetch(). Dikirim dengan Content-Type
   * "text/plain" (BUKAN application/json) supaya browser menganggapnya
   * "simple request" dan tidak memicu preflight OPTIONS, yang biasanya
   * gagal ditangani Apps Script (tidak bisa set header CORS custom di sana).
   */
  function callApi(action, payload) {
    return fetch(API_URL, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify({ token: API_TOKEN, action: action, payload: payload || {} })
    })
      .then(res => {
        if (!res.ok) throw new Error('HTTP ' + res.status);
        return res.json();
      });
  }

  let scannedIdAlat = null;
  let currentScanner = null;
  let daftarAlatProject = []; // [{idAlat, namaAlat}]
  let idTransaksiDicari = null;

  function pilihMode(mode) {
    document.getElementById('mode-select').classList.add('hidden');
    if (mode === 'pinjam') {
      document.getElementById('form-pinjam').classList.remove('hidden');
      startScanner('reader-pinjam', onScanPinjam);
    } else if (mode === 'pinjam-project') {
      document.getElementById('form-pinjam-project').classList.remove('hidden');
      startScanner('reader-project', onScanProject, true);
    } else {
      document.getElementById('form-kembali').classList.remove('hidden');
      startScanner('reader-kembali', onScanKembali);
    }
  }

  function kembaliKeMenu() {
    if (currentScanner) {
      currentScanner.stop().catch(() => {});
      currentScanner = null;
    }
    ['form-pinjam', 'form-pinjam-project', 'form-kembali'].forEach(id => document.getElementById(id).classList.add('hidden'));
    document.getElementById('mode-select').classList.remove('hidden');

    scannedIdAlat = null;
    daftarAlatProject = [];
    idTransaksiDicari = null;
    lastScanTimeProject = 0;
    ['alat-info-pinjam', 'alat-info-kembali', 'hasil-pinjam', 'hasil-kembali',
     'scan-info-project', 'daftar-alat-project', 'hasil-pinjam-project', 'daftar-transaksi-project'
    ].forEach(id => document.getElementById(id).innerHTML = '');
    document.getElementById('jumlah-alat-project').innerText = '0';
    document.getElementById('btn-submit-pinjam').disabled = true;
    document.getElementById('btn-submit-project').disabled = true;
    document.getElementById('btn-submit-kembali').disabled = true;
  }

  function startScanner(elementId, onDecoded, continuous) {
    currentScanner = new Html5Qrcode(elementId);
    currentScanner.start(
      { facingMode: "environment" },
      { fps: 10, qrbox: 220 },
      (decodedText) => {
        if (continuous) {
          onDecoded(decodedText.trim());
        } else {
          currentScanner.stop().catch(() => {});
          onDecoded(decodedText.trim());
        }
      },
      () => { /* ignore scan-frame errors */ }
    ).catch(() => {
      document.getElementById(elementId).innerHTML =
        '<div class="info-box error">Tidak bisa akses kamera. Pastikan izin kamera diaktifkan di browser.</div>';
    });
  }

  // ---------- MODE PINJAM (INDIVIDUAL) ----------
  function onScanPinjam(idAlat) {
    scannedIdAlat = idAlat;
    callApi('getAlatInfo', { idAlat: idAlat })
      .then(showAlatInfoPinjam)
      .catch(err => {
        document.getElementById('alat-info-pinjam').innerHTML =
          '<div class="info-box error">Gagal menghubungi server: ' + err.message + '</div>';
      });
  }

  function showAlatInfoPinjam(info) {
    const div = document.getElementById('alat-info-pinjam');
    if (!info.found) {
      div.innerHTML = '<div class="info-box error">Alat dengan kode "' + scannedIdAlat + '" tidak ditemukan.</div>';
      return;
    }
    if (info.status !== 'Tersedia') {
      div.innerHTML = '<div class="info-box error">' + info.namaAlat + ' sedang tidak tersedia (status: ' + info.status + ').</div>';
      return;
    }
    div.innerHTML = '<div class="info-box">✅ Alat terdeteksi: <b>' + info.namaAlat + '</b> (' + scannedIdAlat + ')</div>';
    document.getElementById('btn-submit-pinjam').disabled = false;
  }

  function submitPinjam() {
    const nama = document.getElementById('nama').value;
    const email = document.getElementById('email').value;
    const departemen = document.getElementById('departemen').value;
    const rencanaKembali = document.getElementById('tanggalKembali').value;

    if (!nama || !email || !rencanaKembali) {
      document.getElementById('hasil-pinjam').innerHTML =
        '<div class="status-msg error">Mohon lengkapi nama, email, dan tanggal rencana kembali.</div>';
      return;
    }

    document.getElementById('btn-submit-pinjam').disabled = true;
    document.getElementById('btn-submit-pinjam').innerText = 'Memproses...';

    callApi('prosesPeminjaman', {
      nama: nama, email: email, departemen: departemen,
      idAlat: scannedIdAlat, rencanaKembali: rencanaKembali
    }).then(onSuccessPinjam).catch(onErrorPinjam);
  }

  function onSuccessPinjam(result) {
    document.getElementById('btn-submit-pinjam').innerText = 'Konfirmasi Peminjaman';
    if (result.success) {
      document.getElementById('hasil-pinjam').innerHTML =
        '<div class="status-msg success">✅ Berhasil! ID Transaksi: <b>' + result.idTransaksi + '</b><br>Konfirmasi juga dikirim ke email.</div>';
    } else {
      document.getElementById('btn-submit-pinjam').disabled = false;
      document.getElementById('hasil-pinjam').innerHTML = '<div class="status-msg error">' + result.message + '</div>';
    }
  }

  function onErrorPinjam(err) {
    document.getElementById('btn-submit-pinjam').disabled = false;
    document.getElementById('btn-submit-pinjam').innerText = 'Konfirmasi Peminjaman';
    document.getElementById('hasil-pinjam').innerHTML = '<div class="status-msg error">Terjadi error: ' + err.message + '</div>';
  }

  // ---------- MODE PINJAM PROJECT (BATCH) ----------
  let sedangCekAlatProject = false;
  let lastScanTimeProject = 0;
  const JEDA_SCAN_MS = 1500; // jeda minimum antar scan sukses, cegah spam saat kamera masih lihat QR yang sama

  function onScanProject(idAlat) {
    const now = Date.now();
    if (now - lastScanTimeProject < JEDA_SCAN_MS) return;
    if (sedangCekAlatProject) return;
    if (daftarAlatProject.some(a => a.idAlat === idAlat)) {
      lastScanTimeProject = now;
      document.getElementById('scan-info-project').innerHTML =
        '<div class="info-box error">Alat "' + idAlat + '" sudah ada di daftar.</div>';
      return;
    }
    sedangCekAlatProject = true;
    callApi('getAlatInfo', { idAlat: idAlat })
      .then(info => {
        sedangCekAlatProject = false;
        lastScanTimeProject = Date.now();
        handleScanProjectResult(idAlat, info);
      })
      .catch(err => {
        sedangCekAlatProject = false;
        document.getElementById('scan-info-project').innerHTML =
          '<div class="info-box error">Gagal menghubungi server: ' + err.message + '</div>';
      });
  }

  function handleScanProjectResult(idAlat, info) {
    const div = document.getElementById('scan-info-project');
    if (!info.found) {
      div.innerHTML = '<div class="info-box error">Alat "' + idAlat + '" tidak ditemukan.</div>';
      return;
    }
    if (info.status !== 'Tersedia') {
      div.innerHTML = '<div class="info-box error">' + info.namaAlat + ' sedang tidak tersedia (status: ' + info.status + ').</div>';
      return;
    }
    daftarAlatProject.push({ idAlat: idAlat, namaAlat: info.namaAlat });
    div.innerHTML = '<div class="info-box">✅ Ditambahkan: <b>' + info.namaAlat + '</b> (' + idAlat + '). Scan alat berikutnya, atau lanjut isi form di bawah.</div>';
    renderDaftarAlatProject();
  }

  function renderDaftarAlatProject() {
    const ul = document.getElementById('daftar-alat-project');
    ul.innerHTML = daftarAlatProject.map((a, idx) =>
      '<li>' + a.namaAlat + ' (' + a.idAlat + ')<button onclick="hapusAlatProject(' + idx + ')">✕</button></li>'
    ).join('');
    document.getElementById('jumlah-alat-project').innerText = daftarAlatProject.length;
    checkFormProjectValid();
  }

  function hapusAlatProject(idx) {
    daftarAlatProject.splice(idx, 1);
    renderDaftarAlatProject();
  }

  function checkFormProjectValid() {
    const nama = document.getElementById('p-nama').value;
    const email = document.getElementById('p-email').value;
    const tanggal = document.getElementById('p-tanggalKembali').value;
    document.getElementById('btn-submit-project').disabled =
      !(nama && email && tanggal && daftarAlatProject.length > 0);
  }
  document.getElementById('p-nama').addEventListener('input', checkFormProjectValid);
  document.getElementById('p-email').addEventListener('input', checkFormProjectValid);
  document.getElementById('p-tanggalKembali').addEventListener('input', checkFormProjectValid);

  function submitPinjamProject() {
    const nama = document.getElementById('p-nama').value;
    const email = document.getElementById('p-email').value;
    const departemen = document.getElementById('p-departemen').value;

          '<div class="info-box error">Gagal menghubungi server: ' + err.message + '</div>';
      });
  }

  function showDaftarTransaksiProject(hasil) {
    const div = document.getElementById('daftar-transaksi-project');
    if (!hasil.found) {
      div.innerHTML = '<div class="info-box error">ID Transaksi tidak ditemukan atau semua alat sudah dikembalikan.</div>';
      return;
    }
    let html = '<div class="info-box">Ditemukan ' + hasil.daftar.length + ' alat:</div>';
    html += '<ul class="item-list">' + hasil.daftar.map(a => '<li>' + a.namaAlat + ' (' + a.idAlat + ')</li>').join('') + '</ul>';
    html += '<label>Kondisi Semua Alat Saat Kembali</label>';
    html += '<select id="k-kondisi-project"><option value="Baik">Baik</option><option value="Perlu Cek">Perlu Cek</option><option value="Rusak">Rusak</option></select>';
    html += '<button class="submit-btn" onclick="submitKembaliProject(' + hasil.daftar.length + ')">Kembalikan Semua (' + hasil.daftar.length + ')</button>';
    html += '<div id="hasil-kembali-project"></div>';
    div.innerHTML = html;
  }

  function submitKembaliProject(jumlah) {
    const kondisi = document.getElementById('k-kondisi-project').value;
    callApi('prosesPengembalianBatch', { idTransaksi: idTransaksiDicari, kondisi: kondisi })
      .then(onSuccessKembaliProject).catch(onErrorKembaliProject);
  }

  function onSuccessKembaliProject(result) {
    const div = document.getElementById('hasil-kembali-project');
    if (result.success) {
      div.innerHTML = '<div class="status-msg success">✅ ' + result.jumlahAlat + ' alat berhasil dikembalikan sekaligus.</div>';
    } else {
      div.innerHTML = '<div class="status-msg error">' + result.message + '</div>';
    }
  }

  function onErrorKembaliProject(err) {
    document.getElementById('hasil-kembali-project').innerHTML = '<div class="status-msg error">Terjadi error: ' + err.message + '</div>';
  }
